TITLE: Moderation Effect Model Template
DATA: FILE = your_data.dat;

VARIABLE: 
  NAMES = {{Outcome}} {{Predictor}} {{Moderator}};
  USEVARIABLES = ALL;
  MISSING = ALL(-999);

DEFINE:
  ! Center variables
  CENTER {{Predictor}} {{Moderator}} (GRANDMEAN);
  ! Create interaction term
  {{Interaction}} = {{Predictor}} * {{Moderator}};

ANALYSIS:
  ESTIMATOR = MLR;
  
MODEL:
  ! Main effects
  {{Outcome}} ON {{Predictor}} {{Moderator}};
  
  ! Interaction effect (moderation effect)
  {{Outcome}} ON {{Interaction}};

OUTPUT:
  STDYX;
  CINTERVAL;

PLOT:
  TYPE = PLOT2;
