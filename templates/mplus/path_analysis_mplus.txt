TITLE: Path Analysis Model Template
DATA: FILE = your_data.dat;

VARIABLE: 
  NAMES = {{Outcome}} {{Predictor1}} {{Predictor2}} {{Predictor3}} {{Mediator}};
  USEVARIABLES = ALL;
  MISSING = ALL(-999);

ANALYSIS:
  ESTIMATOR = MLR;
  
MODEL:
  ! Main paths
  {{Outcome}} ON {{Predictor1}} {{Predictor2}} {{Predictor3}};
  
  ! Mediation paths (optional)
  {{Mediator}} ON {{Predictor1}} {{Predictor2}};
  {{Outcome}} ON {{Mediator}};
  
  ! Exogenous variable covariances
  {{Predictor1}} WITH {{Predictor2}} {{Predictor3}};
  {{Predictor2}} WITH {{Predictor3}};

OUTPUT:
  STDYX;
  CINTERVAL;
