TITLE: Basic SEM Model

VARIABLE:
  USEVARIABLES = {{latent1_item1}} {{latent1_item2}} {{latent1_item3}}
                 {{latent2_item1}} {{latent2_item2}} {{latent2_item3}};
  MISSING ARE ALL (-999);

ANALYSIS:
  TYPE = GENERAL;
  ESTIMATOR = ML;

MODEL:
  ! Measurement model for Latent1
  {{latent1}} BY {{latent1_item1}}* {{latent1_item2}} {{latent1_item3}};
  {{latent1}}@1;
  
  ! Measurement model for Latent2
  {{latent2}} BY {{latent2_item1}}* {{latent2_item2}} {{latent2_item3}};
  {{latent2}}@1;
  
  ! Structural model
  {{latent2}} ON {{latent1}};

OUTPUT:
  STANDARDIZED CINTERVAL MODINDICES;
