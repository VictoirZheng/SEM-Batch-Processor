TITLE: Mediation Model

VARIABLE:
  USEVARIABLES = {{outcome}} {{predictor}} {{mediator}};
  MISSING ARE ALL (-999);

ANALYSIS:
  TYPE = GENERAL;
  ESTIMATOR = ML;
  BOOTSTRAP = 2000;

MODEL:
  ! Path a: Predictor -> Mediator
  {{mediator}} ON {{predictor}} (a);
  
  ! Path b: Mediator -> Outcome
  {{outcome}} ON {{mediator}} (b);
  
  ! Path c': Predictor -> Outcome (direct effect)
  {{outcome}} ON {{predictor}} (cp);

MODEL INDIRECT:
  {{outcome}} IND {{predictor}};

OUTPUT:
  STANDARDIZED CINTERVAL(bcbootstrap);
