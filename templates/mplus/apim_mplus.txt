TITLE: Actor-Partner Interdependence Model (APIM)

VARIABLE:
  USEVARIABLES = {{outcome_actor}} {{outcome_partner}} 
                 {{predictor_actor}} {{predictor_partner}};
  MISSING ARE ALL (-999);

ANALYSIS:
  TYPE = GENERAL;
  ESTIMATOR = ML;

MODEL:
  ! Actor effects (same parameter 'a')
  {{outcome_actor}} ON {{predictor_actor}} (a);
  {{outcome_partner}} ON {{predictor_partner}} (a);
  
  ! Partner effects (same parameter 'p')
  {{outcome_actor}} ON {{predictor_partner}} (p);
  {{outcome_partner}} ON {{predictor_actor}} (p);
  
  ! Correlations
  {{predictor_actor}} WITH {{predictor_partner}};
  {{outcome_actor}} WITH {{outcome_partner}};

OUTPUT:
  STANDARDIZED CINTERVAL;
